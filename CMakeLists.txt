cmake_minimum_required(VERSION 3.24)
project(OutfitGallery VERSION 1.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release)
find_package(spdlog CONFIG REQUIRED)
find_package(directxtk CONFIG REQUIRED)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
if(COMMONLIB_SOURCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(COMMONLIB_PREBUILT OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_SOURCE}" commonlib)
add_library(CommonLib ALIAS CommonLibSSE)
elseif(COMMONLIB_BUNDLE)
add_library(CommonLib STATIC IMPORTED)
set_target_properties(CommonLib PROPERTIES IMPORTED_LOCATION "${COMMONLIB_BUNDLE}/lib/CommonLibSSE.lib")
target_include_directories(CommonLib INTERFACE "${COMMONLIB_BUNDLE}/include" "${COMMONLIB_BUNDLE}/extern/openvr/headers")
target_compile_definitions(CommonLib INTERFACE ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 ENABLE_SKYRIM_VR=1 HAS_SKYRIM_MULTI_TARGETING=1)
target_link_libraries(CommonLib INTERFACE spdlog::spdlog Microsoft::DirectXTK "${COMMONLIB_BUNDLE}/lib/openvr_api.lib" Advapi32 bcrypt D3D11 d3dcompiler Dbghelp DXGI Ole32 Version)
else()
message(FATAL_ERROR "Set COMMONLIB_SOURCE (v4.39.3), or COMMONLIB_BUNDLE built with a compatible MSVC toolset")
endif()
add_library(OutfitGallery SHARED src/main.cpp src/Capture.cpp src/Presets.cpp src/Equipment.cpp src/ManagedItems.cpp)
target_include_directories(OutfitGallery PRIVATE third_party src)
target_include_directories(OutfitGallery SYSTEM PRIVATE "${NLOHMANN_JSON_INCLUDE_DIR}")
target_precompile_headers(OutfitGallery PRIVATE src/PCH.h)
target_compile_options(OutfitGallery PRIVATE /Zc:preprocessor /permissive- /EHsc /utf-8 /W4)
target_compile_definitions(OutfitGallery PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE SMOOTHCAM_API_COMMONLIB)
target_link_libraries(OutfitGallery PRIVATE CommonLib windowscodecs)
set_target_properties(OutfitGallery PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(GALLERY_TESTS "Build software-rendered PNG crop verification" ON)
if(GALLERY_TESTS)
  enable_testing()
  add_executable(CaptureTest tests/CaptureTest.cpp src/Capture.cpp)
  target_include_directories(CaptureTest PRIVATE src)
  target_precompile_headers(CaptureTest PRIVATE src/PCH.h)
  target_compile_options(CaptureTest PRIVATE /Zc:preprocessor /utf-8 /EHsc)
  target_compile_definitions(CaptureTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(CaptureTest PRIVATE CommonLib windowscodecs)
  add_test(NAME PortraitPngCrop COMMAND CaptureTest)
  add_executable(PresetTest tests/PresetTest.cpp src/Presets.cpp)
  target_include_directories(PresetTest PRIVATE src "${NLOHMANN_JSON_INCLUDE_DIR}")
  target_compile_options(PresetTest PRIVATE /utf-8 /EHsc)
  add_test(NAME PresetPersistence COMMAND PresetTest)
endif()
